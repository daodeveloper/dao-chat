# syntax=docker/dockerfile:1.4
FROM python:3.9

WORKDIR /app

ENV PIP_CACHE_DIR=/root/.cache/pip

COPY requirements.txt .

RUN --mount=type=cache,target=/root/.cache/pip \
    pip install --cache-dir=/root/.cache/pip -r requirements.txt

# Pre-download the local embedding model so the first request doesn't wait on it.
RUN python -c "from fastembed import TextEmbedding; TextEmbedding(model_name='BAAI/bge-small-en-v1.5')" || echo "fastembed model will download at runtime"

RUN mkdir -p /app/data/faiss_index

COPY app/ /app/app/
COPY data/ /app/data/

RUN chmod -R 755 /app

CMD ["sh", "-c", "python -m app.build_index_simple ; uvicorn app.main:app --host 0.0.0.0 --port 8080"]
